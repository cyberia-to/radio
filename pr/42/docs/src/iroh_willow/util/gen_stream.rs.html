13/// Wraps a [`Gen`] into a [`Stream`].
14///
15/// The stream yields the items yielded by the generator.
16/// The generator's final output can be retrieved via [`Self::final_output`].
17#[derive(derive_more::Debug)]
18pub struct GenStream<Yield, Err, Fut, FinalOutput = ()>
19where
20    Fut: Future<Output = Result<FinalOutput, Err>>,
21{
22    #[debug("Gen")]
23    gen: Gen<Yield, (), Fut>,
24    is_complete: bool,
25    final_output: Option<FinalOutput>,
26}
27
28impl<Yield, Err, Fut, FinalOutput> GenStream<Yield, Err, Fut, FinalOutput>
29where
30    Fut: Future<Output = Result<FinalOutput, Err>>,
31{
32    pub fn new(producer: impl FnOnce(Co<Yield, ()>) -> Fut) -> Self {
33        Self::from_gen(Gen::new(producer))
34    }
35
36    pub fn from_gen(gen: Gen<Yield, (), Fut>) -> Self {
37        Self {
38            gen,
39            is_complete: false,
40            final_output: None,
41        }
42    }
43
44    pub fn final_output(self) -> Option<FinalOutput> {
45        self.final_output
46    }
47}
48
49impl<Yield, Err, Fut, FinalOutput> Stream for GenStream<Yield, Err, Fut, FinalOutput>
50where
51    Fut: Future<Output = Result<FinalOutput, Err>>,
52    FinalOutput: Unpin,
53{
54    type Item = Result<Yield, Err>;
55
56    fn poll_next(mut self: Pin<&mut Self>, cx: &mut Context<'_>) -> Poll<Option<Self::Item>> {
57        if self.is_complete {
58            return Poll::Ready(None);
59        }
60        let (item, final_output) = {
61            let mut fut = self.gen.async_resume();
62            let out = std::task::ready!(Pin::new(&mut fut).poll(cx));
63            match out {
64                GeneratorState::Yielded(output) => (Some(Ok(output)), None),
65                GeneratorState::Complete(Ok(final_output)) => (None, Some(final_output)),
66                GeneratorState::Complete(Err(err)) => (Some(Err(err)), None),
67            }
68        };
69        if matches!(item, None | Some(Err(_))) {
70            self.is_complete = true;
71        }
72        if let Some(final_output) = final_output {
73            self.final_output = Some(final_output);
74        };
75        Poll::Ready(item)
76    }
77}
