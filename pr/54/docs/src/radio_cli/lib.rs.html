1//! `radio-cli`'s own logic, importable outside the `radio` binary.
2//!
3//! Everything here is free of `clap`, `tokio` and the iroh endpoint: pure
4//! parsing and encoding, the CLI's own domain rules independent of the
5//! terminal it happens to run in. This is the seam a future integration
6//! test (row 22, particle-addressed blob fetch) builds on — an
7//! integration test lives in a separate crate under `tests/`, and a
8//! separate crate can only see a `[lib]` target, not `main.rs`.

10use anyhow::{Context, Result, bail};
11
12/// Decode a hex string into bytes. Non-ASCII input, odd length and non-hex
13/// digits are rejected with a message naming the problem, not a panic.
14pub fn hex_to_bytes(hex: &str) -> Result<Vec<u8>> {
15    if !hex.is_ascii() {
16        bail!("non-ASCII hex string");
17    }
18    if !hex.len().is_multiple_of(2) {
19        bail!("odd-length hex string");
20    }
21    (0..hex.len())
22        .step_by(2)
23        .map(|i| u8::from_str_radix(&hex[i..i + 2], 16).context("invalid hex digit"))
24        .collect()
25}
26
27/// Parse a 64-character hex string into a Poseidon2 hash.
28pub fn parse_poseidon_hash(hex: &str) -> Result<hemera::Hash> {
29    let bytes = hex_to_bytes(hex).context("invalid hex hash")?;
30    if bytes.len() != 32 {
31        bail!("hash must be 32 bytes (64 hex chars), got {} bytes", bytes.len());
32    }
33    let mut arr = [0u8; 32];
34    arr.copy_from_slice(&bytes);
35    Ok(hemera::Hash::from_bytes(arr))
36}

38#[cfg(test)]
39mod tests {
40    use super::*;
41
42    #[test]
43    fn hex_to_bytes_round_trips_lowercase() {
44        assert_eq!(hex_to_bytes("00ff10").unwrap(), vec![0x00, 0xff, 0x10]);
45    }
46
47    #[test]
48    fn hex_to_bytes_rejects_odd_length() {
49        assert!(hex_to_bytes("abc").is_err());
50    }
51
52    #[test]
53    fn hex_to_bytes_rejects_non_hex_digits() {
54        assert!(hex_to_bytes("zz").is_err());
55    }
56
57    #[test]
58    fn hex_to_bytes_rejects_non_ascii_without_panicking() {
59        // 31 ASCII bytes + 'é' (2 bytes) + 1 ASCII byte = 34 bytes, an even
60        // length that used to reach the byte-indexed slice loop and panic
61        // when a step boundary landed inside 'é'.
62        let s = format!("{}é{}", "a".repeat(31), "a");
63        assert_eq!(s.len(), 34);
64        assert!(hex_to_bytes(&s).is_err());
65    }
66
67    #[test]
68    fn hex_to_bytes_empty_is_empty() {
69        assert_eq!(hex_to_bytes("").unwrap(), Vec::<u8>::new());
70    }
71
72    #[test]
73    fn parse_poseidon_hash_round_trips_through_display() {
74        let h = hemera::hash(b"radio-cli lib target");
75        let parsed = parse_poseidon_hash(&h.to_string()).unwrap();
76        assert_eq!(parsed, h);
77    }
78
79    #[test]
80    fn parse_poseidon_hash_rejects_wrong_length() {
81        assert!(parse_poseidon_hash("ab").is_err());
82    }
83
84    #[test]
85    fn parse_poseidon_hash_rejects_bad_hex() {
86        assert!(parse_poseidon_hash(&"z".repeat(64)).is_err());
87    }
