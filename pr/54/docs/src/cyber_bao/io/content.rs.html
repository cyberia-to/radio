1//! Content item types for the BAO streaming protocol.
2
3use bytes::Bytes;
4
5use crate::tree::TreeNode;
6
7/// A parent hash pair in the BAO tree.
8#[derive(Debug, Clone)]
9#[cfg_attr(feature = "serde", derive(serde::Serialize, serde::Deserialize))]
10pub struct Parent<H = hemera::Hash> {
11    /// The tree node this pair belongs to.
12    pub node: TreeNode,
13    /// The (left, right) child hash pair.
14    pub pair: (H, H),
15}
16
17/// A leaf data chunk.
18#[derive(Debug, Clone)]
19#[cfg_attr(feature = "serde", derive(serde::Serialize, serde::Deserialize))]
20pub struct Leaf {
21    /// Byte offset of this leaf in the original data.
22    pub offset: u64,
23    /// The leaf data.
24    pub data: Bytes,
25}
26
27/// A content item yielded during BAO streaming decode.
28///
29/// Either a parent hash pair or a verified leaf data chunk.
30#[derive(Debug, Clone)]
31#[cfg_attr(feature = "serde", derive(serde::Serialize, serde::Deserialize))]
32pub enum BaoContentItem<H = hemera::Hash> {
33    /// A parent hash pair.
34    Parent(Parent<H>),
35    /// A verified leaf data chunk.
36    Leaf(Leaf),
37}
38
39impl<H> BaoContentItem<H> {
40    /// Returns true if this is a leaf item.
41    pub fn is_leaf(&self) -> bool {
42        matches!(self, Self::Leaf(_))
43    }
44
45    /// Returns true if this is a parent item.
46    pub fn is_parent(&self) -> bool {
47        matches!(self, Self::Parent(_))
48    }
49}
