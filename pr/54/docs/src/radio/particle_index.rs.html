1//! Maps a cyber particle hash (flat Hemera hash, the identity `cyb` and the
2//! rest of the graph address a file by) to the iroh-blobs content hash (the
3//! Hemera BAO tree root over 4 KiB leaves, a separate domain) of the same
4//! bytes, so a blob can be requested from a peer by particle instead of by
5//! the transport's own hash. The two hashes are computed independently over
6//! identical bytes and do not coincide, so
7//! resolving "fetch this particle" into "fetch this blob hash" needs this
8//! translation step before the standard `iroh_blobs` get protocol can run.
9//!
10//! Stored as one `<particle-hex> <blob-hash-hex>` pair per line — no new
11//! dependency for a mapping this small.
12
13use std::collections::HashMap;
14use std::path::Path;
15
16use anyhow::{Context, Result};
17
18#[derive(Default, Debug, Clone, PartialEq, Eq)]
19pub struct ParticleIndex(HashMap<String, String>);
20
21impl ParticleIndex {
22    pub fn new() -> Self {
23        Self::default()
24    }
25
26    pub fn insert(&mut self, particle: hemera::Hash, blob_hash: iroh_blobs::Hash) {
27        self.0.insert(particle.to_string(), blob_hash.to_string());
28    }
29
30    pub fn lookup(&self, particle: &hemera::Hash) -> Option<iroh_blobs::Hash> {
31        self.0.get(&particle.to_string())?.parse().ok()
32    }
33
34    #[cfg(test)]
35    pub fn len(&self) -> usize {
36        self.0.len()
37    }
38
39    /// An index file that does not exist yet is an empty index — `Add`
40    /// creates it on first use instead of requiring it be pre-created.
41    pub fn load(path: &Path) -> Result<Self> {
42        if !path.exists() {
43            return Ok(Self::new());
44        }
45        let text = std::fs::read_to_string(path)
46            .with_context(|| format!("reading {}", path.display()))?;
47        let mut map = HashMap::new();
48        for (lineno, line) in text.lines().enumerate() {
49            let line = line.trim();
50            if line.is_empty() {
51                continue;
52            }
53            let mut parts = line.split_whitespace();
54            let particle = parts.next().with_context(|| {
55                format!("{}:{}: missing particle hash", path.display(), lineno + 1)
56            })?;
57            let blob = parts.next().with_context(|| {
58                format!("{}:{}: missing blob hash", path.display(), lineno + 1)
59            })?;
60            map.insert(particle.to_string(), blob.to_string());
61        }
62        Ok(Self(map))
63    }
64
65    pub fn save(&self, path: &Path) -> Result<()> {
66        let mut text = String::with_capacity(self.0.len() * 130);
67        for (particle, blob) in &self.0 {
68            text.push_str(particle);
69            text.push(' ');
70            text.push_str(blob);
71            text.push('\n');
72        }
73        std::fs::write(path, text).with_context(|| format!("writing {}", path.display()))
74    }
75}

77#[cfg(test)]
78mod tests {
79    use super::*;
80
81    fn temp_path(name: &str) -> std::path::PathBuf {
82        std::env::temp_dir().join(format!(
83            "radio-cli-particle-index-test-{}-{}-{}",
84            std::process::id(),
85            name,
86            rand::random::<u32>()
87        ))
88    }
89
90    #[test]
91    fn load_missing_file_is_empty() {
92        let path = temp_path("missing");
93        let idx = ParticleIndex::load(&path).unwrap();
94        assert_eq!(idx.len(), 0);
95    }
96
97    #[test]
98    fn insert_then_lookup_round_trips() {
99        let particle = hemera::hash(b"a bostrom file");
100        let blob_hash = iroh_blobs::Hash::new(b"a bostrom file");
101
102        let mut idx = ParticleIndex::new();
103        idx.insert(particle, blob_hash);
104
105        assert_eq!(idx.lookup(&particle), Some(blob_hash));
106    }
107
108    #[test]
109    fn lookup_unknown_particle_is_none() {
110        let idx = ParticleIndex::new();
111        let particle = hemera::hash(b"never indexed");
112        assert_eq!(idx.lookup(&particle), None);
113    }
114
115    #[test]
116    fn save_then_load_round_trips_across_files() {
117        let path = temp_path("roundtrip");
118        let particle = hemera::hash(b"round trip bytes");
119        let blob_hash = iroh_blobs::Hash::new(b"round trip bytes");
120
121        let mut idx = ParticleIndex::new();
122        idx.insert(particle, blob_hash);
123        idx.save(&path).unwrap();
124
125        let reloaded = ParticleIndex::load(&path).unwrap();
126        assert_eq!(reloaded.lookup(&particle), Some(blob_hash));
127        assert_eq!(reloaded.len(), 1);
128
129        std::fs::remove_file(&path).ok();
130    }
131
132    #[test]
133    fn save_then_load_keeps_multiple_entries() {
134        let path = temp_path("multi");
135        let mut idx = ParticleIndex::new();
136        let entries: Vec<_> = (0..5)
137            .map(|i| {
138                let bytes = format!("entry {i}").into_bytes();
139                let particle = hemera::hash(&bytes);
140                let blob_hash = iroh_blobs::Hash::new(&bytes);
141                idx.insert(particle, blob_hash);
142                (particle, blob_hash)
143            })
144            .collect();
145        idx.save(&path).unwrap();
146
147        let reloaded = ParticleIndex::load(&path).unwrap();
148        assert_eq!(reloaded.len(), 5);
149        for (particle, blob_hash) in entries {
150            assert_eq!(reloaded.lookup(&particle), Some(blob_hash));
151        }
152
153        std::fs::remove_file(&path).ok();
154    }
155
156    #[test]
157    fn blank_lines_are_skipped() {
158        let path = temp_path("blank-lines");
159        let particle = hemera::hash(b"surrounded by blank lines");
160        let blob_hash = iroh_blobs::Hash::new(b"surrounded by blank lines");
161        std::fs::write(&path, format!("\n\n{particle} {blob_hash}\n\n")).unwrap();
162
163        let idx = ParticleIndex::load(&path).unwrap();
164        assert_eq!(idx.lookup(&particle), Some(blob_hash));
165
166        std::fs::remove_file(&path).ok();
167    }
