1//! Incremental fixed-chunk group hashing with a fixed frontier.
2//!
3//! This computes a hash only; it does not authenticate data or a particle/root
4//! association. Each pending step calls one backend kernel, on at most 4096 input
5//! bytes or two child hashes. Arbitrary backend code and hash destructors retain
6//! their own cost and allocation behavior.
7
8use core::fmt;
9
10use crate::{CHUNK_SIZE, HashBackend};
11
12/// A group cannot be represented by the backend's absolute chunk counters.
13#[derive(Debug, Clone, Copy, PartialEq, Eq)]
14pub enum GroupHashError {
15    /// The input length does not fit in a u64 (unreachable on <=64-bit targets).
16    InputTooLarge,
17    /// The last absolute chunk counter would exceed u64::MAX.
18    CounterOverflow,
19}
20
21impl fmt::Display for GroupHashError {
22    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
23        f.write_str(match self {
24            Self::InputTooLarge => "group input length exceeds u64",
25            Self::CounterOverflow => "group chunk counter overflows u64",
26        })
27    }
28}
29
30impl core::error::Error for GroupHashError {}
31
32fn leaf_count(len: u64) -> u64 {
33    (len / CHUNK_SIZE as u64 + u64::from(!len.is_multiple_of(CHUNK_SIZE as u64))).max(1)
34}
35
36fn validate_counter(len: u64, start: u64) -> Result<u64, GroupHashError> {
37    let leaves = leaf_count(len);
38    start
39        .checked_add(leaves - 1)
40        .ok_or(GroupHashError::CounterOverflow)?;
41    Ok(leaves)
42}
43
44/// A resumable complete-left binary reduction over fixed 4096-byte chunks.
45///
46/// Construction and completed polls call no backend method. Every pending
47/// [`step`](Self::step) performs exactly one leaf or parent kernel: for N chunks
48/// (at least one, even for empty input), completion takes exactly 2*N-1 steps.
49/// Only the completing kernel receives the requested root flag.
50///
51/// The caller owns scheduling: check [`is_complete`](Self::is_complete), admit
52/// one unit from its persistent budget, then step. Intermediate steps return
53/// `None` while `is_complete` is false; count kernel entries, not returned hashes.
54/// No internal quota is renewed by polling. Backend panics have no recovery
55/// contract. Hashes are moved, never cloned.
56///
57/// The frontier uses 53 slots: a u64 byte length needs at most 2^52 leaves;
58/// before a leaf there are at most 52 complete peaks, plus the incoming leaf.
59/// This fixed storage does not require allocation or a placeholder hash.
60pub struct GroupHasher<'a, B: HashBackend> {
61    backend: &'a B,
62    remaining: &'a [u8],
63    next_counter: u64,
64    leaves: u64,
65    processed: u64,
66    frontier: [Option<B::Hash>; 53],
67    occupied: usize,
68    reductions: u32,
69    root: bool,
70    complete: bool,
71}
72
73impl<B: HashBackend> fmt::Debug for GroupHasher<'_, B> {
74    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
75        f.debug_struct("GroupHasher")
76            .field("remaining_bytes", &self.remaining.len())
77            .field("processed_leaves", &self.processed)
78            .field("leaves", &self.leaves)
79            .field("occupied", &self.occupied)
80            .field("is_complete", &self.complete)
81            .finish_non_exhaustive()
82    }
83}
84
85impl<'a, B: HashBackend> GroupHasher<'a, B> {
86    /// Borrow a group and validate every counter before any hashing.
87    ///
88    /// Empty input is a single empty leaf. A sole leaf at u64::MAX is valid;
89    /// two leaves starting there return [`GroupHashError::CounterOverflow`].
90    pub fn new(
91        backend: &'a B,
92        data: &'a [u8],
93        start_chunk: u64,
94        is_root: bool,
95    ) -> Result<Self, GroupHashError> {
96        let len = u64::try_from(data.len()).map_err(|_| GroupHashError::InputTooLarge)?;
97        let leaves = validate_counter(len, start_chunk)?;
98        Ok(Self {
99            backend,
100            remaining: data,
101            next_counter: start_chunk,
102            leaves,
103            processed: 0,
104            frontier: core::array::from_fn(|_| None),
105            occupied: 0,
106            reductions: 0,
107            root: is_root,
108            complete: false,
109        })
110    }
111
112    /// Whether the final hash has already been returned.
113    pub fn is_complete(&self) -> bool {
114        self.complete
115    }
116
117    /// Perform one pending kernel and return the hash only on the final step.
118    ///
119    /// Intermediate steps return None with `is_complete() == false`. Completed
120    /// polls return None with `is_complete() == true` and perform no kernel.
121    pub fn step(&mut self) -> Option<B::Hash> {
122        if self.complete {
123            return None;
124        }
125        if self.reductions != 0 {
126            let right = self.frontier[self.occupied - 1]
127                .take()
128                .expect("occupied right peak");
129            let left = self.frontier[self.occupied - 2]
130                .take()
131                .expect("occupied left peak");
132            let final_parent = self.processed == self.leaves && self.occupied == 2;
133            let hash = self
134                .backend
135                .parent_hash(&left, &right, self.root && final_parent);
136            self.occupied -= 1;
137            self.reductions -= 1;
138            if final_parent {
139                self.complete = true;
140                return Some(hash);
141            }
142            self.frontier[self.occupied - 1] = Some(hash);
143        } else {
144            let take = self.remaining.len().min(CHUNK_SIZE);
145            let hash = self.backend.chunk_hash(
146                &self.remaining[..take],
147                self.next_counter,
148                self.root && self.leaves == 1,
149            );
150            self.remaining = &self.remaining[take..];
151            self.processed += 1;
152            if self.leaves == 1 {
153                self.complete = true;
154                return Some(hash);
155            }
156            self.frontier[self.occupied] = Some(hash);
157            self.occupied += 1;
158            if self.processed == self.leaves {
159                // Fold remaining complete peaks from right to left.
160                self.reductions = (self.occupied - 1) as u32;
161            } else {
162                // Merge equal-size complete peaks before accepting another leaf.
163                self.reductions = self.processed.trailing_zeros();
164                self.next_counter += 1; // all used counters were validated in new
165            }
166        }
167        None
168    }
169}
