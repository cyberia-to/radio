1//! Checked, allocation-free preorder geometry for one byte range.
2//!
3//! This cursor emits geometry only. It does not parse a proof or authenticate
4//! a particle, root, payload or empty request.
5
6use core::{fmt, iter::FusedIterator, ops::Range};
7
8use crate::{BaoChunk, BlockSize, CHUNK_SIZE, TreeNode};

10/// Invalid geometry supplied to [`RangeTraversal::new`].
11#[derive(Debug, Clone, Copy, PartialEq, Eq)]
12pub enum RangeTraversalError {
13    /// The group width cannot be represented by both u64 and usize.
14    InvalidBlockSize,
15    /// The byte range is reversed or extends beyond the file.
16    InvalidRange,
17}
18
19impl fmt::Display for RangeTraversalError {
20    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
21        f.write_str(match self {
22            Self::InvalidBlockSize => "invalid traversal block size",
23            Self::InvalidRange => "invalid traversal byte range",
24        })
25    }
26}
27
28impl core::error::Error for RangeTraversalError {}

30#[derive(Debug, Clone, Copy)]
31struct Frame {
32    first: u64,
33    count: u64,
34    is_root: bool,
35}
36
37/// Preorder cursor over whole groups intersecting a half-open byte range.
38///
39/// Construction uses fixed storage for 53 frames. Each pending [`Iterator::next`]
40/// pops one frame, pushes at most two children and returns one item, without
41/// allocation, recursion, scanning or hashing. Pausing requires no extra state:
42/// simply retain the cursor without calling `next`.
43///
44/// ROOT belongs to the complete file tree, even when the request selects only
45/// one leaf. Parent flags identify children visited by this request; a parent
46/// still describes a pair of hashes. An empty request is immediately complete
47/// and emits no item; it makes no authentication claim.
48pub struct RangeTraversal {
49    size: u64,
50    group_bytes: u64,
51    selected: Range<u64>,
52    frames: [Frame; 53],
53    pending: usize,
54}
55
56impl fmt::Debug for RangeTraversal {
57    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
58        f.debug_struct("RangeTraversal")
59            .field("size", &self.size)
60            .field("group_bytes", &self.group_bytes)
61            .field("selected", &self.selected)
62            .field("pending", &self.pending)
63            .finish()
64    }
65}
66
67impl RangeTraversal {
68    /// Admit a file size, group width and half-open requested byte interval.
69    ///
70    /// Block geometry is validated before the request: the block log must be at
71    /// most 51, and its byte width must fit usize. This admits logs 0..=51 on
72    /// 64-bit targets and 0..=19 on 32-bit targets, even for a short or empty file.
73    /// Width admission concerns integer geometry, not allocation layout validity.
74    /// The request must satisfy `start <= end <= size`; an empty interval at EOF
75    /// is valid. Only the final file group is clipped to its actual byte length.
76    pub fn new(
77        size: u64,
78        block_size: BlockSize,
79        request: Range<u64>,
80    ) -> Result<Self, RangeTraversalError> {
81        if block_size.0 > 51 {
82            return Err(RangeTraversalError::InvalidBlockSize);
83        }
84        let group_bytes = (CHUNK_SIZE as u64) << block_size.0;
85        usize::try_from(group_bytes).map_err(|_| RangeTraversalError::InvalidBlockSize)?;
86        if request.start > request.end || request.end > size {
87            return Err(RangeTraversalError::InvalidRange);
88        }
89        let mut result = Self {
90            size,
91            group_bytes,
92            selected: 0..0,
93            frames: [Frame {
94                first: 0,
95                count: 0,
96                is_root: false,
97            }; 53],
98            pending: 0,
99        };
100        if request.start < request.end {
101            result.selected = request.start / group_bytes..(request.end - 1) / group_bytes + 1;
102            result.push(Frame {
103                first: 0,
104                count: size.div_ceil(group_bytes),
105                is_root: true,
106            });
107        }
108        Ok(result)
109    }
110
111    /// Whether no item remains, including immediately after the last emission.
112    pub fn is_complete(&self) -> bool {
113        self.pending == 0
114    }
115
116    fn push(&mut self, frame: Frame) {
117        // N <= 2^52: at depth d <= 52, at most d deferred siblings plus
118        // one current frame exist. Pop before pushing children preserves 53.
119        self.frames[self.pending] = frame;
120        self.pending += 1;
121    }
122}
123
124impl Iterator for RangeTraversal {
125    type Item = BaoChunk;
126
127    fn next(&mut self) -> Option<Self::Item> {
128        if self.is_complete() {
129            return None;
130        }
131        self.pending -= 1;
132        let frame = self.frames[self.pending];
133        if frame.count == 1 {
134            // first < N implies first * group_bytes <= size - 1. Subtract
135            // before clipping: the next group's byte end need not fit u64.
136            let start = frame.first * self.group_bytes;
137            return Some(BaoChunk::Leaf {
138                start_chunk: start / CHUNK_SIZE as u64,
139                size: self.group_bytes.min(self.size - start) as usize,
140                is_root: frame.is_root,
141            });
142        }
143        let left_count = 1u64 << (frame.count - 1).ilog2();
144        let middle = frame.first + left_count;
145        let end = frame.first + frame.count;
146        let left = self.selected.start < middle && frame.first < self.selected.end;
147        let right = self.selected.start < end && middle < self.selected.end;
148        if right {
149            self.push(Frame {
150                first: middle,
151                count: frame.count - left_count,
152                is_root: false,
153            });
154        }
155        if left {
156            self.push(Frame {
157                first: frame.first,
158                count: left_count,
159                is_root: false,
160            });
161        }
162        // middle < N <= 2^52, so this group-based index is below 2^53.
163        Some(BaoChunk::Parent {
164            node: TreeNode::new(2 * middle - 1),
165            is_root: frame.is_root,
166            left,
167            right,
168        })
169    }
170}
