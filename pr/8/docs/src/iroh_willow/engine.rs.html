35/// The [`Engine`] is the main handle onto a Willow store with networking.
36///
37/// It runs a dedicated thread for all storage operations, and a peer manager to coordinate network
38/// connections to other peers.
39///
40/// The engine does not establish any peer connections on its own. Synchronisation sessions can be
41/// started with [`Engine::sync_with_peer`].
42#[derive(Debug, Clone)]
43pub struct Engine {
44    actor_handle: ActorHandle,
45    pub(crate) endpoint: Endpoint,
46    peer_manager_inbox: mpsc::Sender<peer_manager::Input>,
47    // `Engine` needs to be `Clone + Send`, and we need to `task.await` in its `shutdown()` impl.
48    // So we need
49    // - `Shared` so we can `task.await` from all `Node` clones
50    // - `MapErr` to map the `JoinError` to a `String`, because `JoinError` is `!Clone`
51    // - `AbortOnDropHandle` to make sure that the `task` is cancelled when all `Node`s are dropped
52    //   (`Shared` acts like an `Arc` around its inner future).
53    peer_manager_task: Shared<MapErr<AbortOnDropHandle<Result<(), String>>, JoinErrToStr>>,
54    rpc_handler: Arc<OnceLock<crate::rpc::handler::RpcHandler>>,
55}
56
57pub(crate) type JoinErrToStr = Box<dyn Fn(JoinError) -> String + Send + Sync + 'static>;
58
59impl Engine {
60    /// Get an in memory client to interact with the willow engine.
61    pub fn client(&self) -> &MemClient {
62        &self
63            .rpc_handler
64            .get_or_init(|| RpcHandler::new(self.clone()))
65            .client
66    }
67
68    /// Start the Willow engine.
69    ///
70    /// This needs an `endpoint` to connect to other peers, and a `create_store` closure which
71    /// returns a [`Storage`] instance.
72    ///
73    /// You also need to pass [`AcceptOpts`] to configure what to do with incoming connections.
74    /// Its default implementation will accept all connections and run sync with all our interests.
75    ///
76    /// To actually accept connections, an [`Endpoint::accept`] loop has to be run outside of the
77    /// engine, passing all connections that match [`crate::net::ALPN`] to the engine with
78    /// [`Engine::handle_connection`].
79    ///
80    /// The engine will spawn a dedicated storage thread, and the `create_store` closure will be called on
81    /// this thread, so that the [`Storage`] does not have to be `Send`.
82    pub fn spawn<S: Storage>(
83        endpoint: Endpoint,
84        create_store: impl 'static + Send + FnOnce() -> S,
85        accept_opts: AcceptOpts,
86    ) -> Self {
87        let me = endpoint.id();
88        let actor_handle = ActorHandle::spawn(create_store, me);
89        let (pm_inbox_tx, pm_inbox_rx) = mpsc::channel(PEER_MANAGER_INBOX_CAP);
90        let peer_manager = PeerManager::new(
91            actor_handle.clone(),
92            endpoint.clone(),
93            pm_inbox_rx,
94            accept_opts,
95        );
96        let peer_manager_task = tokio::task::spawn(
97            async move { peer_manager.run().await.map_err(|e| e.to_string()) }
98                .instrument(error_span!("peer_manager", me=%me.fmt_short())),
99        );
100        let peer_manager_task = AbortOnDropHandle::new(peer_manager_task)
101            .map_err(Box::new(|e: JoinError| e.to_string()) as JoinErrToStr)
102            .shared();
103        Engine {
104            actor_handle,
105            endpoint,
106            peer_manager_inbox: pm_inbox_tx,
107            peer_manager_task,
108            rpc_handler: Default::default(),
109        }
110    }
111
112    /// Handle an incoming connection.
113    pub async fn handle_connection(&self, conn: Connection) -> Result<()> {
114        self.peer_manager_inbox
115            .send(peer_manager::Input::HandleConnection { conn })
116            .await?;
117        Ok(())
118    }
119
120    /// Synchronises with a peer.
121    ///
122    /// Will try to establish a connection to `peer` if there is none already, and then open a
123    /// synchronisation session.
124    ///
125    /// `init` contains the initialisation options for this synchronisation intent.
126    ///
127    /// Returns an [`IntentHandle`] which receives events and can submit updates into the session.
128    ///
129    /// This can freely be called multiple times for the same peer. The engine will merge the
130    /// intents and make sure that only a single session is opened per peer.
131    pub async fn sync_with_peer(&self, peer: EndpointId, init: SessionInit) -> Result<IntentHandle> {
132        let (intent, handle) = Intent::new(init);
133        self.peer_manager_inbox
134            .send(peer_manager::Input::SubmitIntent { peer, intent })
135            .await?;
136        Ok(handle)
137    }
138
139    /// Shutdown the engine.
140    ///
141    /// This will try to close all connections gracefully for up to 10 seconds,
142    /// and abort them otherwise.
143    pub async fn shutdown(&self) -> Result<()> {
144        debug!("shutdown engine");
145        let (reply, reply_rx) = oneshot::channel();
146        self.peer_manager_inbox
147            .send(peer_manager::Input::Shutdown { reply })
148            .await?;
149        reply_rx.await?;
150        let res = self.peer_manager_task.clone().await;
151        match res {
152            Err(err) => error!(?err, "peer manager task panicked"),
153            Ok(Err(err)) => error!(?err, "peer manager task failed"),
154            Ok(Ok(())) => {}
155        };
156        debug!("shutdown engine: peer manager terminated");
157        self.actor_handle.shutdown().await?;
158        debug!("shutdown engine: willow actor terminated");
159        Ok(())
160    }
161}
162
163impl std::ops::Deref for Engine {
164    type Target = ActorHandle;
165
166    fn deref(&self) -> &Self::Target {
167        &self.actor_handle
168    }
169}
170
171impl ProtocolHandler for Engine {
172    async fn accept(&self, conn: Connection) -> Result<(), AcceptError> {
173        self.handle_connection(conn).await.map_err(|e| AcceptError::from(std::io::Error::other(e)))
174    }
175
176    async fn shutdown(&self) {
177        crate::engine::Engine::shutdown(self).await.ok();
178    }
