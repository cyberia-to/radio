71/// Establish the connection by running the initial transmission and
72/// opening the streams for the control and logical channels.
73///
74/// The initial transmission is transferred over a pair of uni streams.
75/// All channels for the actual WGPS are bi streams.
76/// Returns the initial transmission and [`ChannelStreams], which is an
77/// array of send and receive streams, one for each WGPS channel.
78///
79/// To start the networking loops that pipe the QUIC streams into our
80/// internal channel streams use [`prepare_channels`].
81pub(crate) async fn establish(
82    conn: &Connection,
83    our_role: Role,
84    our_nonce: AccessChallenge,
85) -> Result<(InitialTransmission, ChannelStreams)> {
86    debug!(?our_role, "establishing connection");
87    // Run the initial transmission (which uses uni streams) concurrently
88    // with opening/accepting the bi streams for the channels.
89    let fut = (
90        initial_transmission(conn, our_nonce),
91        open_channel_streams(conn, our_role),
92    )
93        .try_join();
94    tokio::time::timeout(ESTABLISH_TIMEOUT, fut).await?
95}

97async fn initial_transmission(
98    conn: &Connection,
99    our_nonce: AccessChallenge,
100) -> Result<InitialTransmission> {
101    let challenge_hash = our_nonce.hash();
102    let mut send_stream = conn.open_uni().await?;
103    send_stream.write_u8(MAX_PAYLOAD_SIZE_POWER).await?;
104    send_stream.write_all(challenge_hash.as_bytes()).await?;
105
106    let mut recv_stream = conn.accept_uni().await?;
107
108    let their_max_payload_size = {
109        let power = recv_stream.read_u8().await?;
110        ensure!(power <= 64, "max payload size too large");
111        2u64.pow(power as u32)
112    };
113
114    let mut received_commitment = [0u8; CHALLENGE_HASH_LENGTH];
115    recv_stream.read_exact(&mut received_commitment).await?;
116    debug!("initial transmission complete");
117    Ok(InitialTransmission {
118        our_nonce,
119        received_commitment: ChallengeHash::from_bytes(received_commitment),
120        their_max_payload_size,
121    })
122}

130async fn open_channel_streams(conn: &Connection, our_role: Role) -> Result<ChannelStreams> {
131    let channels = match our_role {
132        // Alfie opens a quic stream for each logical channel, and sends a single byte with the
133        // channel id.
134        Role::Alfie => {
135            Channel::all()
136                .map(|channel| {
137                    let conn = conn.clone();
138                    async move {
139                        let (mut send, recv) = conn.open_bi().await?;
140                        send.write_u8(channel.id()).await?;
141                        trace!(?channel, "opened bi stream");
142                        Ok::<_, anyhow::Error>((channel, send, recv))
143                    }
144                })
145                .try_join()
146                .await
147        }
148        // Betty accepts as many quick streams as there are logical channels, and reads a single
149        // byte on each, which is expected to contain a channel id.
150        Role::Betty => {
151            Channel::all()
152                .map(|_| async {
153                    let (send, mut recv) = conn.accept_bi().await?;
154                    let channel_id = recv.read_u8().await?;
155                    let channel = Channel::from_id(channel_id)?;
156                    trace!(?channel, "accepted bi stream");
157                    Result::Ok((channel, send, recv))
158                })
159                .try_join()
160                .await
161        }
162    }?;
163    Ok(channels)
164}

166/// Create a future for each WGPS channel that pipes between the QUIC channels and the
167/// [`Sender`] and [`Receiver`] for each channel to be used in the session.
168///
169/// Returns [`Channels`], which contains all senders and receivers, and a future that drives
170/// the send and receive loops for all channels combined.
171pub(crate) fn prepare_channels(
172    channels: ChannelStreams,
173) -> Result<(Channels, impl Future<Output = Result<()>> + Send)> {
174    let mut channels = channels.map(|(ch, send, recv)| (ch, Some(prepare_channel(ch, send, recv))));
175
176    let mut find = |channel| {
177        channels
178            .iter_mut()
179            .find_map(|(ch, streams)| (*ch == channel).then(|| streams.take()))
180            .flatten()
181            .ok_or(MissingChannel(channel))
182    };
183
184    let ctrl = find(Channel::Control)?;
185    let pai = find(Channel::Logical(LogicalChannel::Intersection))?;
186    let rec = find(Channel::Logical(LogicalChannel::Reconciliation))?;
187    let stt = find(Channel::Logical(LogicalChannel::StaticToken))?;
188    let aoi = find(Channel::Logical(LogicalChannel::AreaOfInterest))?;
189    let cap = find(Channel::Logical(LogicalChannel::Capability))?;
190    let dat = find(Channel::Logical(LogicalChannel::Data))?;
191
192    let fut = async move {
193        let result = (ctrl.2, pai.2, rec.2, stt.2, aoi.2, cap.2, dat.2)
194            .try_join()
195            .map_ok(|_| ())
196            .await;
197        if let Err(err) = &result {
198            debug!("channels closed with error: {err:#}");
199        } else {
200            debug!("channels closed");
201        }
202        result
203    };
204
205    let channels = Channels {
206        send: ChannelSenders {
207            control_send: ctrl.0,
208            logical_send: LogicalChannelSenders {
209                intersection_send: pai.0,
210                reconciliation_send: rec.0,
211                static_tokens_send: stt.0,
212                aoi_send: aoi.0,
213                capability_send: cap.0,
214                data_send: dat.0,
215            },
216        },
217        recv: ChannelReceivers {
218            control_recv: ctrl.1,
219            logical_recv: LogicalChannelReceivers {
220                intersection_recv: pai.1.into(),
221                reconciliation_recv: rec.1.into(),
222                static_tokens_recv: stt.1.into(),
223                aoi_recv: aoi.1.into(),
224                capability_recv: cap.1.into(),
225                data_recv: dat.1.into(),
226            },
227        },
228    };
229    Ok((channels, fut))
230}

232fn prepare_channel(
233    ch: Channel,
234    send_stream: SendStream,
235    recv_stream: RecvStream,
236) -> (
237    Sender<Message>,
238    Receiver<Message>,
239    impl Future<Output = Result<()>> + Send,
240) {
241    let guarantees = match ch {
242        Channel::Control => Guarantees::Unlimited,
243        Channel::Logical(_) => Guarantees::Limited(0),
244    };
245    let cap = CHANNEL_CAP;
246    let (sender, outbound_reader) = outbound_channel(cap, guarantees);
247    let (inbound_writer, receiver) = inbound_channel(cap);
248
249    let recv_fut = recv_loop(ch, recv_stream, inbound_writer)
250        .map_err(move |e| e.context(format!("receive loop for {ch:?} failed")));
251
252    let send_fut = send_loop(ch, send_stream, outbound_reader)
253        .map_err(move |e| e.context(format!("send loop for {ch:?} failed")));
254
255    let fut = (recv_fut, send_fut).try_join().map_ok(|_| ());
256
257    (sender, receiver, fut)
258}

260/// Error code when stopping receive streams because we closed our session.
261///
262/// This is currently for debugging purposes only - the other end will still see this as a connection error.
263const ERROR_CODE_SESSION_CLOSED: VarInt = VarInt::from_u32(1);
264
265async fn recv_loop(
266    channel: Channel,
267    mut recv_stream: RecvStream,
268    mut channel_writer: Writer,
269) -> Result<()> {
270    trace!(?channel, "recv: start");
271    let max_buffer_size = channel_writer.max_buffer_size();
272    while let Some(buf) = recv_stream
273        .read_chunk(max_buffer_size)
274        .await
275        .context("failed to read from quic stream")?
276    {
277        trace!(len = buf.bytes.len(), "read");
278        match channel_writer.write_all(&buf.bytes[..]).await {
279            Ok(()) => {
280                trace!(len = buf.bytes.len(), "sent");
281            }
282            Err(err) if err.kind() == io::ErrorKind::BrokenPipe => {
283                debug!("closing recv channel: session closed");
284                recv_stream.stop(ERROR_CODE_SESSION_CLOSED)?;
285                break;
286            }
287            Err(err) => return Err(err.into()),
288        }
289    }
290    trace!(?channel, "recv: stream close");
291    channel_writer.close();
292    Ok(())
293}

295async fn send_loop(
296    channel: Channel,
297    mut send_stream: SendStream,
298    channel_reader: Reader,
299) -> Result<()> {
300    trace!(?channel, "send: start");
301    while let Some(data) = channel_reader.read_bytes().await {
302        // let len = data.len();
303        // trace!(len, "send");
304        send_stream
305            .write_chunk(data)
306            .await
307            .context("failed to write to quic stream")?;
308        // trace!(len, "sent");
309    }
310    trace!(?channel, "send: close writer");
311    send_stream.finish()?;
312    send_stream.stopped().await?;
313    // We don't await SendStream::stopped, because we rely on application level closing notifications,
314    // and make sure that the connection is closed gracefully in any case.
315    trace!(?channel, "send: done");
316    Ok(())
317}

319/// Terminates a connection gracefully.
320///
321/// This function should be called after all bidirectional streams are terminated (`finish` called
322/// for send streams and `read_to_end` awaited for recv stream) and no further streams will be
323/// opened or accepted by the application.
324///
325/// It will send a goodbye byte over a newly opened uni channel, and wait for the other peer to
326/// confirm either by sending a goodbye byte as well or closing the connection with
327/// [`ERROR_CODE_OK`], signalling that our goodbye byte was received.
328///
329/// We will only close the connection after having received the goodbye byte, or after the other
330/// peer closed the connection.
331///
332/// This flow guarantees that neither peer will close the connection too early.
333///
334/// A connection is considered to be closed gracefully if and only if this procedure is run to end
335/// successfully, and if the connection is closed with the expected error code.
336///
337/// Returns an error if the termination flow was aborted prematurely or if the connection was not
338/// closed with the expected error code.
339pub(crate) async fn terminate_gracefully(conn: Connection) -> Result<()> {
340    trace!("terminating connection");
341    // Send a single byte on a newly opened uni stream.
342    let mut send_stream = conn.open_uni().await?;
343    send_stream.write_u8(1).await?;
344    send_stream.finish()?;
345    // Wait until we either receive the goodbye byte from the other peer, or for the other peer
346    // to close the connection with the expected error code.
347    match tokio::time::timeout(SHUTDOWN_TIMEOUT, wait_for_goodbye_or_graceful_close(&conn)).await {
348        Ok(Ok(())) => {
349            conn.close(ERROR_CODE_OK, b"bye");
350            trace!("connection terminated gracefully");
351            Ok(())
352        }
353        Ok(Err(err)) => {
354            conn.close(ERROR_CODE_FAIL, b"failed-while-closing");
355            trace!(?err, "connection failed while terminating");
356            Err(err)
357        }
358        Err(err) => {
359            conn.close(ERROR_CODE_FAIL, b"timeout-while-closing");
360            trace!("connection timed out while terminating");
361            Err(err.into())
362        }
363    }
364}

366/// Waits for a goodbye byte or connection close, and then closes the connection.
367///
368/// Accepts a single uni stream and reads a single byte on it.
369///
370/// Returns once we received the goodbye byte or if the peer closed the connection with the
371/// graceful error code.
372///
373/// Returns an error if the connection was closed without the graceful error code or if reading the
374/// goodbye byte failed.
375async fn wait_for_goodbye_or_graceful_close(conn: &Connection) -> Result<()> {
376    let mut recv_stream = match conn.accept_uni().await {
377        // The other peer closed the connection with the expected error code: They received our
378        // goodbye byte after having sent theirs. We're free to close the connection.
379        Err(ConnectionError::ApplicationClosed(frame)) if frame.error_code == ERROR_CODE_OK => {
380            return Ok(())
381        }
382        // The peer closed the connection with an unexpected error coe.
383        Err(err) => return Err(err.into()),
384        Ok(stream) => stream,
385    };
386    let mut buf = [0u8];
387    match recv_stream.read_exact(&mut buf).await {
388        // We received the goodbye byte: the other peer indicates to us that they are finished with
389        // everything and we are free to close the connection.
390        Ok(()) if buf == [1u8] => Ok(()),
391        // The other peer closed the connection with the expected error code: They received our
392        // goodbye byte, and reacted by closing the connection. We're free to close too.
393        Err(ReadExactError::ReadError(ReadError::ConnectionLost(
394            ConnectionError::ApplicationClosed(frame),
395        ))) if frame.error_code == ERROR_CODE_OK => Ok(()),
396        // The peer has sent invalid data on the goodbye stream.
397        Ok(()) => Err(anyhow!("Received unexpected closing byte from peer.")),
398        // The peer closed the connection with an unexpected error coe.
399        Err(err) => Err(err.into()),
400    }
401}

615        let mut rng = create_rng("net_live_data");

617        let disco = iroh::address_lookup::memory::MemoryLookup::new();
618        let (ep_alfie, node_id_alfie, _) = create_endpoint(&mut rng, &disco).await?;
619        let (ep_betty, node_id_betty, addr_betty) = create_endpoint(&mut rng, &disco).await?;
620
621        let handle_alfie = ActorHandle::spawn_memory(iroh_blobs::store::mem::MemStore::default().into(), node_id_alfie);
622        let handle_betty = ActorHandle::spawn_memory(iroh_blobs::store::mem::MemStore::default().into(), node_id_betty);
623
624        let user_alfie = handle_alfie.create_user().await?;
625        let user_betty = handle_betty.create_user().await?;
626
627        let namespace_id = handle_alfie
628            .create_namespace(NamespaceKind::Owned, user_alfie)
629            .await?;
630
631        let cap_for_betty = handle_alfie
632            .delegate_caps(
633                CapSelector::any(namespace_id),
634                AccessMode::Write,
635                DelegateTo::new(user_betty, RestrictArea::None),
636            )
637            .await?;
638
639        handle_betty.import_caps(cap_for_betty).await?;
640
641        let mut expected_entries = BTreeSet::new();
642        let start = Instant::now();
643
644        let n_init = 2;
645        insert(
646            &handle_alfie,
647            namespace_id,
648            user_alfie,
649            n_init,
650            |n| Path::from_bytes(&[b"alfie-init", n.to_string().as_bytes()]),
651            |n| format!("alfie{n}"),
652            &mut expected_entries,
653        )
654        .await?;
655
656        insert(
657            &handle_betty,
658            namespace_id,
659            user_betty,
660            n_init,
661            |n| Path::from_bytes(&[b"betty-init", n.to_string().as_bytes()]),
662            |n| format!("betty{n}"),
663            &mut expected_entries,
664        )
665        .await?;
666
667        info!("init took {:?}", start.elapsed());
668
669        let start = Instant::now();
670        let (conn_alfie, conn_betty) = tokio::join!(
671            async { ep_alfie.connect(addr_betty, ALPN).await.unwrap() },
672            async { ep_betty.accept().await.unwrap().await.unwrap() }
673        );
674        info!("connecting took {:?}", start.elapsed());
675
676        let start = Instant::now();
677        let (done_tx, done_rx) = tokio::sync::oneshot::channel();
678
679        // alfie insert 3 entries after waiting a second
680        let _insert_task_alfie = tokio::task::spawn({
681            let handle_alfie = handle_alfie.clone();
682            let count = 3;
683            let content_fn = |i: usize| format!("alfie live {i}");
684            let path_fn = |i: usize| Path::from_bytes(&[b"alfie-live", i.to_string().as_bytes()]);
685            let mut track_entries = vec![];
686
687            async move {
688                tokio::time::sleep(std::time::Duration::from_secs(1)).await;
689                insert(
690                    &handle_alfie,
691                    namespace_id,
692                    user_alfie,
693                    count,
694                    path_fn,
695                    content_fn,
696                    &mut track_entries,
697                )
698                .await
699                .expect("failed to insert");
700                done_tx.send(track_entries).unwrap();
701            }
702        });
703
704        let init_alfie = SessionInit::new(Interests::All, SessionMode::Continuous);
705        let init_betty = SessionInit::new(Interests::All, SessionMode::Continuous);
706
707        let (intent_alfie, intent_handle_alfie) = Intent::new(init_alfie);
708        let (intent_betty, mut intent_handle_betty) = Intent::new(init_betty);
709
710        let nonce_alfie = AccessChallenge::generate_with_rng(&mut rng);
711        let nonce_betty = AccessChallenge::generate_with_rng(&mut rng);
712
713        let (session_alfie, session_betty) = tokio::join!(
714            run(
715                node_id_alfie,
716                handle_alfie.clone(),
717                conn_alfie.clone(),
718                Role::Alfie,
719                nonce_alfie,
720                vec![intent_alfie]
721            ),
722            run(
723                node_id_betty,
724                handle_betty.clone(),
725                conn_betty.clone(),
726                Role::Betty,
727                nonce_betty,
728                vec![intent_betty]
729            )
730        );
731        let (mut session_alfie, net_task_alfie) = session_alfie?;
732        let (mut session_betty, net_task_betty) = session_betty?;
733
734        let live_entries = done_rx.await?;
735        expected_entries.extend(live_entries);
736        // TODO: replace with event
737        tokio::time::sleep(Duration::from_secs(1)).await;
738
739        // Drop the alfie intent, which closes the session.
740        drop(intent_handle_alfie);
741
742        let (senders_alfie, _alfie_cancelled) = session_alfie
743            .complete()
744            .await
745            .expect("failed to close alfie session");
746        info!("close alfie session");
747        senders_alfie.close_all();
748
749        let (senders_betty, _betty_cancelled) = session_betty
750            .complete()
751            .await
752            .expect("failed to close alfie session");
753        senders_betty.close_all();
754
755        let (r1, r2) = tokio::try_join!(net_task_alfie, net_task_betty)
756            .expect("failed to close connection loops");
757        r1.unwrap();
758        r2.unwrap();
759
760        let res_betty = intent_handle_betty.complete().await;
761        info!(time=?start.elapsed(), "finished");
762        info!("betty intent res {:?}", res_betty);
763        assert!(res_betty.is_err());
764        assert_eq!(
765            res_betty,
766            Err(Arc::new(crate::session::Error::SessionClosedByPeer))
767        );
768
769        tokio::try_join!(
770            terminate_gracefully(conn_alfie),
771            terminate_gracefully(conn_betty),
772        )
773        .expect("failed to close both connections gracefully");
774
775        // Close endpoints gracefully
776        ep_alfie.close().await;
777        ep_betty.close().await;
778
779        let alfie_entries = get_entries(&handle_alfie, namespace_id).await?;
780        let betty_entries = get_entries(&handle_betty, namespace_id).await?;
781        info!("alfie has now {} entries", alfie_entries.len());
782        info!("betty has now {} entries", betty_entries.len());
783        // not using assert_eq because it would print a lot in case of failure
784        assert!(alfie_entries == expected_entries, "alfie expected entries");
785        assert!(betty_entries == expected_entries, "betty expected entries");
786
787        Ok(())
788    }

790    /// Generate an iroh SecretKey from a rand 0.8 rng.
791    /// This bridges the rand_core 0.6 / 0.9 version gap.
792    fn generate_secret_key(rng: &mut impl rand_core::CryptoRngCore) -> SecretKey {
793        let bytes: [u8; 32] = rand::Rng::gen(rng);
794        SecretKey::from_bytes(&bytes)
795    }

797    pub async fn create_endpoint(
798        rng: &mut rand_chacha::ChaCha12Rng,
799        disco: &iroh::address_lookup::memory::MemoryLookup,
800    ) -> Result<(Endpoint, EndpointId, EndpointAddr)> {
801        let ep = Endpoint::empty_builder(iroh::RelayMode::Disabled)
802            .secret_key(generate_secret_key(rng))
803            .address_lookup(disco.clone())
804            .alpns(vec![ALPN.to_vec()])
805            .bind()
806            .await?;
807        let addr = ep.addr();
808        let node_id = ep.id();
809        disco.add_endpoint_info(addr.clone());
810        Ok((ep, node_id, addr))
811    }

813    async fn get_entries(store: &ActorHandle, namespace: NamespaceId) -> Result<BTreeSet<Entry>> {
814        let entries: Result<BTreeSet<_>> = store
815            .get_entries(namespace, Range3d::new_full())
816            .await?
817            .map(|entry| entry.map(|entry| entry.into_parts().0))
818            .try_collect()
819            .await;
820        entries
821    }

823    async fn insert(
824        handle: &ActorHandle,
825        namespace_id: NamespaceId,
826        user_id: UserId,
827        count: usize,
828        path_fn: impl Fn(usize) -> Result<Path, InvalidPathError2>,
829        content_fn: impl Fn(usize) -> String,
830        track_entries: &mut impl Extend<Entry>,
831    ) -> Result<()> {
832        for i in 0..count {
833            let payload = content_fn(i).as_bytes().to_vec();
834            let path = path_fn(i).expect("invalid path");
835            let entry = EntryForm {
836                namespace_id,
837                subspace_id: SubspaceForm::User,
838                path,
839                timestamp: TimestampForm::Now,
840                payload: PayloadForm::Bytes(payload.into()),
841            };
842            let (entry, inserted) = handle.insert_entry(entry, AuthForm::Any(user_id)).await?;
843            assert!(inserted);
844            track_entries.extend([entry.into_parts().0]);
845        }
846        Ok(())
847    }

849    fn parse_env_var<T>(var: &str, default: T) -> T
850    where
851        T: std::str::FromStr,
852        T::Err: std::fmt::Debug,
853    {
854        match std::env::var(var).as_deref() {
855            Ok(val) => val
856                .parse()
857                .unwrap_or_else(|_| panic!("failed to parse environment variable {var}")),
858            Err(_) => default,
859        }
860    }
