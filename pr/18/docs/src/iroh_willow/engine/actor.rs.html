49    pub fn spawn<S: Storage>(
50        create_store: impl 'static + Send + FnOnce() -> S,
51        me: EndpointId,
52    ) -> ActorHandle {
53        let (inbox_tx, inbox_rx) = tokio::sync::mpsc::channel(INBOX_CAP);
54        let join_handle = std::thread::Builder::new()
55            .name("willow".to_string())
56            .spawn(move || {
57                let span = error_span!("willow", me=%me.fmt_short());
58                let _guard = span.enter();
59                let store = Store::new((create_store)());
60                let actor = Actor::new(store, inbox_rx);
61                if let Err(error) = actor.run() {
62                    error!(?error, "willow actor failed");
63                };
64            })
65            .expect("failed to spawn willow-actor thread");
66        let join_handle = Arc::new(Some(join_handle));
67        ActorHandle {
68            inbox_tx,
69            join_handle,
70        }
71    }

255impl Drop for ActorHandle {
256    fn drop(&mut self) {
257        // this means we're dropping the last reference
258        if let Some(handle) = Arc::get_mut(&mut self.join_handle) {
259            // gain ownership of handle
260            let handle = handle.take().expect("can only drop once");
261
262            // gain ownership of inbox_tx
263            let (dumb, _) = tokio::sync::mpsc::channel(1);
264            let inbox_tx = std::mem::replace(&mut self.inbox_tx, dumb);
265
266            // shutdown
267            let shutdown = move || {
268                inbox_tx.blocking_send(Input::Shutdown { reply: None }).ok();
269                if let Err(err) = handle.join() {
270                    warn!(?err, "Failed to join sync actor");
271                }
272            };
273
274            match tokio::runtime::Handle::try_current() {
275                Ok(runtime) => {
276                    // We shouldn't block the runtime
277                    runtime.spawn_blocking(shutdown);
278                }
279                Err(_) => {
280                    // We can do everything sync
281                    shutdown();
282                }
283            }
284        }
285    }

362#[derive(Debug)]
363struct Actor<S: Storage> {
364    inbox_rx: tokio::sync::mpsc::Receiver<Input>,
365    store: Store<S>,
366    next_session_id: u64,
367    tasks: JoinSet<()>,
368}
369
370impl<S: Storage> Actor<S> {
371    pub fn new(store: Store<S>, inbox_rx: tokio::sync::mpsc::Receiver<Input>) -> Self {
372        Self {
373            store,
374            inbox_rx,
375            next_session_id: 0,
376            tasks: Default::default(),
377        }
378    }
379
380    pub fn run(self) -> Result<()> {
381        let rt = tokio::runtime::Builder::new_current_thread()
382            .build()
383            .expect("failed to start current-thread runtime for willow actor");
384        let local_set = tokio::task::LocalSet::new();
385        local_set.block_on(&rt, async move { self.run_async().await })
386    }
387
388    async fn run_async(mut self) -> Result<()> {
389        loop {
390            tokio::select! {
391                msg = self.inbox_rx.recv() => match msg {
392                    None => break,
393                    Some(Input::Shutdown { reply }) => {
394                        self.tasks.shutdown().await;
395                        drop(self);
396                        if let Some(reply) = reply {
397                            reply.send(()).ok();
398                        }
399                        break;
400                    }
401                    Some(msg) => {
402                        if self.handle_message(msg).await.is_err() {
403                            warn!("failed to send reply: receiver dropped");
404                        }
405                    }
406                },
407            };
408        }
409        Ok(())
410    }
411
412    fn next_session_id(&mut self) -> u64 {
413        let id = self.next_session_id;
414        self.next_session_id += 1;
415        id
416    }
417
418    async fn handle_message(&mut self, message: Input) -> Result<(), SendReplyError> {
419        trace!(%message, "tick: handle_message");
420        match message {
421            Input::Shutdown { .. } => unreachable!("handled in run"),
422            Input::InitSession {
423                conn,
424                intents,
425                reply,
426            } => {
427                let session_id = self.next_session_id();
428                let store = self.store.clone();
429
430                let (update_tx, update_rx) = mpsc::channel(SESSION_UPDATE_CHANNEL_CAP);
431                let (event_tx, event_rx) = mpsc::channel(SESSION_EVENT_CHANNEL_CAP);
432                let update_rx = tokio_stream::wrappers::ReceiverStream::new(update_rx);
433
434                let peer = conn.peer;
435                let future = run_session(
436                    store,
437                    conn,
438                    intents,
439                    session_id,
440                    EventSender(event_tx),
441                    update_rx,
442                )
443                .instrument(error_span!("session", peer = %peer.fmt_short()));
444
445                self.tasks.spawn_local(async move {
446                    if let Err(err) = future.await {
447                        debug!(?peer, ?session_id, ?err, "session failed");
448                    }
449                });
450
451                let handle = SessionHandle {
452                    update_tx,
453                    event_rx,
454                };
455                send_reply(reply, Ok(handle))
456            }
457            Input::GetEntries {
458                namespace,
459                range,
460                reply,
461            } => {
462                let snapshot = self.store.entries().snapshot();
463                match snapshot {
464                    Err(err) => reply.send(Err(err)).await.map_err(send_reply_error),
465                    Ok(snapshot) => {
466                        self.tasks.spawn_local(async move {
467                            match snapshot.get_authorised_entries(namespace, &range) {
468                                Ok(iter) => {
469                                    for entry in iter {
470                                        if reply.send(entry).await.is_err() {
471                                            break;
472                                        }
473                                    }
474                                }
475                                Err(err) => {
476                                    let _ = reply.send(Err(err)).await;
477                                }
478                            }
479                        });
480                        Ok(())
481                    }
482                }
483            }
484            Input::GetEntry {
485                namespace,
486                subspace,
487                path,
488                reply,
489            } => {
490                let res = self
491                    .store
492                    .entries()
493                    .reader()
494                    .get_entry(namespace, subspace, &path);
495                send_reply(reply, res)
496            }
497            Input::IngestEntry {
498                authorised_entry,
499                origin,
500                reply,
501            } => {
502                let res = self.store.entries().ingest_entry(&authorised_entry, origin);
503                send_reply(reply, res)
504            }
505            Input::InsertEntry { entry, auth, reply } => {
506                let res = self.store.insert_entry(entry, auth).await;
507                let res = res.map_err(Into::into);
508                send_reply(reply, res)
509            }
510            Input::InsertSecret { secret, reply } => {
511                let res = self.store.secrets().insert(secret);
512                send_reply(reply, res.map_err(anyhow::Error::from))
513            }
514            Input::CreateNamespace { kind, owner, reply } => {
515                let res = self
516                    .store
517                    .create_namespace(&mut rand::thread_rng(), kind, owner);
518                send_reply(reply, res.map_err(anyhow::Error::from))
519            }
520            Input::CreateUser { reply } => {
521                let secret = UserSecretKey::generate(&mut rand::thread_rng());
522                let res = self.store.secrets().insert_user(secret);
523                send_reply(reply, res.map_err(anyhow::Error::from))
524            }
525            Input::ImportCaps { caps, reply } => {
526                let res = self.store.auth().import_caps(caps);
527                send_reply(reply, res.map_err(anyhow::Error::from))
528            }
529            Input::DelegateCaps {
530                from,
531                access_mode,
532                to,
533                store,
534                reply,
535            } => {
536                let res = self
537                    .store
538                    .auth()
539                    .delegate_full_caps(from, access_mode, to, store);
540                send_reply(reply, res.map_err(anyhow::Error::from))
541            }
542            Input::ResolveInterests { interests, reply } => {
543                let res = self.store.auth().resolve_interests(interests);
544                send_reply(reply, res.map_err(anyhow::Error::from))
545            }
546            Input::SubscribeArea {
547                namespace,
548                area,
549                params,
550                sender,
551            } => {
552                let store = self.store.clone();
553                self.tasks.spawn_local(async move {
554                    // TODO: We wouldn't need to manually forward in a loop here if subscribe_area took a sender
555                    // instead of returning a stream.
556                    let mut stream = store.entries().subscribe_area(namespace, area, params);
557                    while let Some(event) = stream.next().await {
558                        if sender.send(event).await.is_err() {
559                            break;
560                        }
561                    }
562                });
563                Ok(())
564            }
565            Input::ResumeSubscription {
566                progress_id,
567                namespace,
568                area,
569                params,
570                sender,
571            } => {
572                let store = self.store.clone();
573                self.tasks.spawn_local(async move {
574                    let mut stream =
575                        store
576                            .entries()
577                            .resume_subscription(progress_id, namespace, area, params);
578                    while let Some(event) = stream.next().await {
579                        if sender.send(event).await.is_err() {
580                            break;
581                        }
582                    }
583                });
584                Ok(())
585            }
586        }
587    }
588}

591struct SendReplyError;
592
593fn send_reply<T>(sender: oneshot::Sender<T>, value: T) -> Result<(), SendReplyError> {
594    sender.send(value).map_err(send_reply_error)
595}

597fn send_reply_error<T>(_err: T) -> SendReplyError {
598    SendReplyError
