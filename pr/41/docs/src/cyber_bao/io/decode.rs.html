1//! Streaming decode with verification.
2//!
3//! The decoder reads a combined-encoded stream (header + pre-order tree)
4//! and verifies every hash against the trusted root. Invalid data is
5//! rejected immediately.
6
7use crate::hash::HashBackend;
8use crate::tree::{BaoChunk, BaoTree, BlockSize};
9
10/// Error during decoding / verification.
11#[derive(Debug, Clone, PartialEq, Eq)]
12pub enum DecodeError {
13    /// The encoded data is truncated or malformed.
14    Truncated,
15    /// A parent hash pair didn't match the expected parent CV.
16    ParentMismatch { node: u64 },
17    /// A leaf chunk's hash didn't match the expected CV.
18    LeafMismatch { start_chunk: u64 },
19    /// The declared size in the header doesn't match actual data.
20    SizeMismatch { declared: u64, actual: u64 },
21}
22
23impl std::fmt::Display for DecodeError {
24    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
25        match self {
26            DecodeError::Truncated => write!(f, "truncated or malformed encoded data"),
27            DecodeError::ParentMismatch { node } => {
28                write!(f, "parent hash mismatch at node {node}")
29            }
30            DecodeError::LeafMismatch { start_chunk } => {
31                write!(f, "leaf hash mismatch at chunk {start_chunk}")
32            }
33            DecodeError::SizeMismatch { declared, actual } => {
34                write!(f, "size mismatch: declared {declared}, actual {actual}")
35            }
36        }
37    }
38}
39
40impl std::error::Error for DecodeError {}
41
42/// Decode and verify a combined-encoded blob.
43///
44/// Takes the encoded bytes and the trusted root hash. Returns the
45/// original data on success or a `DecodeError` on verification failure.
46pub fn decode<B: HashBackend>(
47    backend: &B,
48    encoded: &[u8],
49    root_hash: &B::Hash,
50    block_size: BlockSize,
51) -> Result<Vec<u8>, DecodeError> {
52    if encoded.len() < 8 {
53        return Err(DecodeError::Truncated);
54    }
55
56    let declared_size = u64::from_le_bytes(encoded[..8].try_into().unwrap());
57    let tree = BaoTree::new(declared_size, block_size);
58    let pre_order = tree.pre_order_chunks();
59    let hash_size = backend.hash_size();
60
61    let mut cursor = 8usize;
62    let mut data = Vec::with_capacity(declared_size as usize);
63
64    // Stack of expected hashes. Push root first; as we encounter parent nodes,
65    // we verify them and push their children (right first, so left is popped first).
66    let mut expected_stack: Vec<B::Hash> = vec![root_hash.clone()];
67
68    for chunk in &pre_order {
69        match chunk {
70            BaoChunk::Parent { node, is_root, .. } => {
71                let pair_size = hash_size * 2;
72                if cursor + pair_size > encoded.len() {
73                    return Err(DecodeError::Truncated);
74                }
75
76                let left_bytes = &encoded[cursor..cursor + hash_size];
77                let right_bytes = &encoded[cursor + hash_size..cursor + pair_size];
78                cursor += pair_size;
79
80                let left_hash = backend.hash_from_bytes(left_bytes);
81                let right_hash = backend.hash_from_bytes(right_bytes);
82
83                // Verify: parent_hash(left, right) must equal expected
84                let computed_parent = backend.parent_hash(&left_hash, &right_hash, *is_root);
85                let expected = expected_stack.pop().ok_or(DecodeError::Truncated)?;
86                if computed_parent != expected {
87                    return Err(DecodeError::ParentMismatch { node: node.0 });
88                }
89
90                // Push children onto expected stack: right first (popped after left)
91                let blocks = tree.blocks();
92                if let Some(right_child) = node.right_child() {
93                    let right_block_start = right_child.chunk_range().start.0 / 2;
94                    if right_block_start < blocks {
95                        expected_stack.push(right_hash);
96                    }
97                }
98                expected_stack.push(left_hash);
99            }
100            BaoChunk::Leaf {
101                start_chunk,
102                size,
103                is_root,
104            } => {
105                if cursor + *size > encoded.len() {
106                    return Err(DecodeError::Truncated);
107                }
108
109                let leaf_data = &encoded[cursor..cursor + *size];
110                cursor += *size;
111
112                let computed = super::hash_group(backend, leaf_data, *start_chunk, *is_root);
113                let expected = expected_stack.pop().ok_or(DecodeError::Truncated)?;
114                if computed != expected {
115                    return Err(DecodeError::LeafMismatch {
116                        start_chunk: *start_chunk,
117                    });
118                }
119
120                data.extend_from_slice(leaf_data);
121            }
122        }
123    }
124
125    if data.len() as u64 != declared_size {
126        return Err(DecodeError::SizeMismatch {
127            declared: declared_size,
128            actual: data.len() as u64,
129        });
130    }
131
132    Ok(data)
133}

142    #[test]
143    fn decode_single_block() {
144        let backend = Poseidon2Backend;
145        let data = vec![0x42u8; CHUNK_SIZE];
146        let (root, encoded) = encode::encode(&backend, &data, BlockSize::ZERO);
147        let decoded = decode(&backend, &encoded, &root, BlockSize::ZERO).unwrap();
148        assert_eq!(decoded, data);
149    }

219    #[test]
220    fn roundtrip_various_sizes() {
221        let backend = Poseidon2Backend;
222        for size in [0, 1, 512, CHUNK_SIZE, CHUNK_SIZE + 1, CHUNK_SIZE * 2, CHUNK_SIZE * 3 - 1000, CHUNK_SIZE * 4, CHUNK_SIZE * 5 - 1000, CHUNK_SIZE * 8] {
223            let data: Vec<u8> = (0..size).map(|i| (i % 256) as u8).collect();
224            let (root, encoded) = encode::encode(&backend, &data, BlockSize::ZERO);
225            let decoded = decode(&backend, &encoded, &root, BlockSize::ZERO).unwrap();
226            assert_eq!(decoded, data, "roundtrip failed for size {size}");
227        }
228    }
